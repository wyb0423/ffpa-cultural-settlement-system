# Root = country
ffcs_initialize_country_schema_v1 = {
	if = {
		limit = { NOT = { has_variable = ffcs_settlement_schema_v1 } }
		set_variable = { name = ffcs_settlement_schema_v1 value = 1 }
	}
}

# Root = sponsor country
ffcs_decrement_active_settlement_count_v1 = {
	if = {
		limit = {
			has_variable = ffcs_active_settlement_count_v1
			var:ffcs_active_settlement_count_v1 > 1
		}
		change_variable = { name = ffcs_active_settlement_count_v1 add = -1 }
	}
	else = {
		remove_variable = ffcs_active_settlement_count_v1
	}
}

# Root = target owner country
ffcs_decrement_inbound_settlement_count_v1 = {
	if = {
		limit = {
			has_variable = ffcs_inbound_settlement_count_v1
			var:ffcs_inbound_settlement_count_v1 > 1
		}
		change_variable = { name = ffcs_inbound_settlement_count_v1 add = -1 }
	}
	else = {
		remove_variable = ffcs_inbound_settlement_count_v1
	}
}

# Root = state carrying a project
ffcs_clear_project_variables_v1 = {
	remove_modifier = ffcs_cultural_settlement_project
	state_region = { remove_variable = ffcs_settlement_sponsor_v1 }
	remove_variable = ffcs_internal_transfer_guard_v1
	remove_variable = ffcs_settlement_sponsor_v1
	remove_variable = ffcs_settlement_original_owner_v1
	remove_variable = ffcs_settlement_progress_v1
	remove_variable = ffcs_settlement_phase_v1
	remove_variable = ffcs_settlement_resistance_v1
	remove_variable = ffcs_settlement_route_v2
	remove_variable = ffcs_transfer_budget_v2
	clear_variable_list = ffcs_settlement_provinces_v2
}

# Root = state region; ffcs_settlement_sponsor temporary scope must exist
ffcs_ensure_overseas_port_v2 = {
	every_scope_state = {
		limit = { owner = scope:ffcs_settlement_sponsor }
		if = {
			limit = { NOT = { has_building = building_port } }
			create_building = {
				building = building_port
				level = 1
			}
		}
	}
}

# Root = state carrying a project
ffcs_cancel_settlement_project_v1 = {
	save_temporary_scope_as = ffcs_settlement_state
	state_region = { save_temporary_scope_as = ffcs_settlement_state_region }
	var:ffcs_settlement_sponsor_v1 ?= {
		save_temporary_scope_as = ffcs_settlement_sponsor
		remove_list_variable = {
			name = ffcs_active_settlement_states_v1
			target = scope:ffcs_settlement_state
		}
		ffcs_decrement_active_settlement_count_v1 = yes
	}
	var:ffcs_settlement_original_owner_v1 ?= {
		ffcs_decrement_inbound_settlement_count_v1 = yes
	}
	ffcs_clear_project_variables_v1 = yes
	debug_log = "FFCS|PROJECT_CANCELLED|DATE=[TimeKeeper.GetCurrentDate.GetString]|SPONSOR=[SCOPE.sCountry('ffcs_settlement_sponsor').GetNameNoFormatting]|REGION=[SCOPE.sStateRegion('ffcs_settlement_state_region').GetNameNoFormatting]|PROJECT=[SCOPE.sState('ffcs_settlement_state').GetNameNoFormatting]|LOCK=CLEARED"
	scope:ffcs_settlement_sponsor ?= {
		post_notification = ffcs_settlement_cancelled
	}
}

# Root = target-owner state; DIVISOR = remaining phase count;
# ffcs_settlement_project = state carrying the project variables
ffcs_apply_settlement_phase_v2 = {
	state_region = { save_temporary_scope_as = ffcs_settlement_state_region }
	set_local_variable = {
		name = ffcs_phase_transfer_budget_v2
		value = {
			value = num_provinces
			subtract = 1
			divide = $DIVISOR$
			ceiling = yes
			min = 0
		}
	}
	scope:ffcs_settlement_project = {
		set_variable = {
			name = ffcs_transfer_budget_v2
			value = local_var:ffcs_phase_transfer_budget_v2
		}
	}

	if = {
		limit = {
			scope:ffcs_settlement_project = {
				NOT = { has_variable_list = ffcs_settlement_provinces_v2 }
			}
		}
		if = {
			limit = {
				ffcs_generated_has_land_seed_v2 = {
					COUNTRY = scope:ffcs_settlement_sponsor
					TARGET = scope:ffcs_settlement_original_owner
				}
			}
			scope:ffcs_settlement_project = {
				set_variable = { name = ffcs_settlement_route_v2 value = 1 }
			}
			ffcs_generated_take_land_seed_v2 = {
				COUNTRY = scope:ffcs_settlement_sponsor
				TARGET = scope:ffcs_settlement_original_owner
				PROJECT = scope:ffcs_settlement_project
			}
		}
		else_if = {
			limit = {
				ffcs_generated_has_port_seed_v2 = {
					TARGET = scope:ffcs_settlement_original_owner
				}
			}
			scope:ffcs_settlement_project = {
				set_variable = { name = ffcs_settlement_route_v2 value = 2 }
			}
			ffcs_generated_take_port_seed_v2 = {
				COUNTRY = scope:ffcs_settlement_sponsor
				TARGET = scope:ffcs_settlement_original_owner
				PROJECT = scope:ffcs_settlement_project
			}
		}

		if = {
			limit = {
				scope:ffcs_settlement_project = {
					has_variable_list = ffcs_settlement_provinces_v2
				}
			}
			if = {
				limit = {
					scope:ffcs_settlement_project = {
						var:ffcs_settlement_route_v2 = 2
					}
				}
				state_region = { ffcs_ensure_overseas_port_v2 = yes }
			}
			if = {
				limit = { has_global_variable = ffcs_debug_enabled_v1 }
				debug_log = "FFCS|SEED_TRANSFERRED|DATE=[TimeKeeper.GetCurrentDate.GetString]|SPONSOR=[SCOPE.sCountry('ffcs_settlement_sponsor').GetNameNoFormatting]|STATE=[THIS.GetState.GetNameNoFormatting]"
			}
			save_temporary_scope_as = ffcs_settlement_state
			scope:ffcs_settlement_sponsor = {
				post_notification = ffcs_settlement_foothold
			}
		}
	}

	while = {
		limit = {
			scope:ffcs_settlement_project = {
				var:ffcs_transfer_budget_v2 > 0
				has_variable_list = ffcs_settlement_provinces_v2
			}
			num_provinces > 1
			ffcs_generated_has_frontier_v2 = {
				COUNTRY = scope:ffcs_settlement_sponsor
				TARGET = scope:ffcs_settlement_original_owner
				PROJECT = scope:ffcs_settlement_project
			}
		}
		ffcs_generated_transfer_frontier_sweep_v2 = {
			COUNTRY = scope:ffcs_settlement_sponsor
			TARGET = scope:ffcs_settlement_original_owner
			PROJECT = scope:ffcs_settlement_project
		}
	}
	scope:ffcs_settlement_project = {
		remove_variable = ffcs_transfer_budget_v2
	}
	debug_log = "FFCS|PHASE_SWEEP_FINISHED|DATE=[TimeKeeper.GetCurrentDate.GetString]|SPONSOR=[SCOPE.sCountry('ffcs_settlement_sponsor').GetNameNoFormatting]|REGION=[SCOPE.sStateRegion('ffcs_settlement_state_region').GetNameNoFormatting]|TARGET=[THIS.GetState.GetNameNoFormatting]"
}

# Root = target-owner state; ffcs_settlement_project = project state
ffcs_complete_settlement_project_v1 = {
	save_temporary_scope_as = ffcs_settlement_state
	state_region = { save_temporary_scope_as = ffcs_settlement_state_region }
	scope:ffcs_settlement_project = {
		var:ffcs_settlement_sponsor_v1 ?= {
			save_temporary_scope_as = ffcs_settlement_sponsor
		}
		var:ffcs_settlement_original_owner_v1 ?= {
			save_temporary_scope_as = ffcs_settlement_original_owner
		}
	}
	set_local_variable = {
		name = ffcs_completion_route_v2
		value = scope:ffcs_settlement_project.var:ffcs_settlement_route_v2
	}
	set_local_variable = {
		name = ffcs_completion_transfer_budget_v2
		value = num_provinces
	}
	scope:ffcs_settlement_project = {
		set_variable = {
			name = ffcs_transfer_budget_v2
			value = local_var:ffcs_completion_transfer_budget_v2
		}
		set_variable = ffcs_internal_transfer_guard_v1
	}
	while = {
		limit = {
			scope:ffcs_settlement_project = {
				var:ffcs_transfer_budget_v2 > 0
				has_variable_list = ffcs_settlement_provinces_v2
			}
			num_provinces > 1
			ffcs_generated_has_frontier_v2 = {
				COUNTRY = scope:ffcs_settlement_sponsor
				TARGET = scope:ffcs_settlement_original_owner
				PROJECT = scope:ffcs_settlement_project
			}
		}
		ffcs_generated_transfer_frontier_sweep_v2 = {
			COUNTRY = scope:ffcs_settlement_sponsor
			TARGET = scope:ffcs_settlement_original_owner
			PROJECT = scope:ffcs_settlement_project
		}
	}
	scope:ffcs_settlement_project = {
		remove_variable = ffcs_transfer_budget_v2
	}

	set_local_variable = { name = ffcs_claim_final_state_v2 value = 0 }
	if = {
		limit = {
			num_provinces = 1
			OR = {
				AND = {
					scope:ffcs_settlement_project = {
						has_variable_list = ffcs_settlement_provinces_v2
					}
					ffcs_generated_has_frontier_v2 = {
						COUNTRY = scope:ffcs_settlement_sponsor
						TARGET = scope:ffcs_settlement_original_owner
						PROJECT = scope:ffcs_settlement_project
					}
				}
				AND = {
					scope:ffcs_settlement_project = {
						NOT = { has_variable_list = ffcs_settlement_provinces_v2 }
						var:ffcs_settlement_route_v2 = 1
					}
					ffcs_generated_has_land_seed_v2 = {
						COUNTRY = scope:ffcs_settlement_sponsor
						TARGET = scope:ffcs_settlement_original_owner
					}
				}
				AND = {
					scope:ffcs_settlement_project = {
						NOT = { has_variable_list = ffcs_settlement_provinces_v2 }
						var:ffcs_settlement_route_v2 = 2
					}
					is_coastal = yes
					ffcs_generated_has_port_seed_v2 = {
						TARGET = scope:ffcs_settlement_original_owner
					}
				}
			}
		}
		set_local_variable = { name = ffcs_claim_final_state_v2 value = 1 }
	}

	scope:ffcs_settlement_sponsor = {
		remove_list_variable = {
			name = ffcs_active_settlement_states_v1
			target = scope:ffcs_settlement_project
		}
		ffcs_decrement_active_settlement_count_v1 = yes
	}
	scope:ffcs_settlement_original_owner = {
		ffcs_decrement_inbound_settlement_count_v1 = yes
	}
	scope:ffcs_settlement_project = {
		ffcs_clear_project_variables_v1 = yes
	}

	if = {
		limit = { local_var:ffcs_claim_final_state_v2 = 1 }
		set_state_owner = scope:ffcs_settlement_sponsor
	}
	scope:ffcs_settlement_state_region = {
		every_scope_state = {
			limit = {
				owner = scope:ffcs_settlement_sponsor
				is_incorporated = no
			}
			set_state_type = unincorporated
		}
		if = {
			limit = { local_var:ffcs_completion_route_v2 = 2 }
			ffcs_ensure_overseas_port_v2 = yes
		}
	}
	debug_log = "FFCS|PROJECT_COMPLETED|DATE=[TimeKeeper.GetCurrentDate.GetString]|SPONSOR=[SCOPE.sCountry('ffcs_settlement_sponsor').GetNameNoFormatting]|REGION=[SCOPE.sStateRegion('ffcs_settlement_state_region').GetNameNoFormatting]|RESULT=COMPLETED_CONTIGUOUS|LOCK=CLEARED"
	scope:ffcs_settlement_sponsor = {
		post_notification = ffcs_settlement_completed
	}
}

# Root = target-owner state; ffcs_settlement_project = project state
ffcs_apply_monthly_settlement_progress_v1 = {
	save_temporary_scope_as = ffcs_settlement_state
	scope:ffcs_settlement_project = {
		var:ffcs_settlement_sponsor_v1 ?= {
			save_temporary_scope_as = ffcs_settlement_sponsor
		}
		var:ffcs_settlement_original_owner_v1 ?= {
			save_temporary_scope_as = ffcs_settlement_original_owner
		}

		change_variable = {
			name = ffcs_settlement_progress_v1
			add = ffcs_monthly_settlement_progress_value
		}

		change_variable = { name = ffcs_settlement_resistance_v1 add = 1 }
	}

	if = {
		limit = {
			scope:ffcs_settlement_project = {
				var:ffcs_settlement_resistance_v1 >= 40
			}
		}
		random_list = {
			5 = {
				scope:ffcs_settlement_project = {
					change_variable = { name = ffcs_settlement_progress_v1 add = -10 }
				}
				scope:ffcs_settlement_sponsor = {
					post_notification = ffcs_settlement_setback
				}
			}
			95 = { }
		}
	}

	if = {
		limit = {
			scope:ffcs_settlement_project = {
				var:ffcs_settlement_progress_v1 < 0
			}
		}
		scope:ffcs_settlement_project = {
			set_variable = { name = ffcs_settlement_progress_v1 value = 0 }
		}
	}

	if = {
		limit = {
			scope:ffcs_settlement_project = {
				var:ffcs_settlement_resistance_v1 >= 100
			}
		}
		scope:ffcs_settlement_project = {
			ffcs_cancel_settlement_project_v1 = yes
		}
	}
	else_if = {
		limit = {
			scope:ffcs_settlement_project = {
				var:ffcs_settlement_progress_v1 >= 25
				var:ffcs_settlement_phase_v1 < 1
			}
		}
		scope:ffcs_settlement_project = { set_variable = ffcs_internal_transfer_guard_v1 }
		ffcs_apply_settlement_phase_v2 = { DIVISOR = 4 }
		scope:ffcs_settlement_project = {
			remove_variable = ffcs_internal_transfer_guard_v1
			set_variable = { name = ffcs_settlement_phase_v1 value = 1 }
		}
	}
	if = {
		limit = {
			scope:ffcs_settlement_project = {
				var:ffcs_settlement_progress_v1 >= 50
				var:ffcs_settlement_phase_v1 < 2
			}
		}
		scope:ffcs_settlement_project = { set_variable = ffcs_internal_transfer_guard_v1 }
		ffcs_apply_settlement_phase_v2 = { DIVISOR = 3 }
		scope:ffcs_settlement_project = {
			remove_variable = ffcs_internal_transfer_guard_v1
			set_variable = { name = ffcs_settlement_phase_v1 value = 2 }
		}
		scope:ffcs_settlement_sponsor = {
			post_notification = ffcs_settlement_phase_2
		}
	}
	if = {
		limit = {
			scope:ffcs_settlement_project = {
				var:ffcs_settlement_progress_v1 >= 75
				var:ffcs_settlement_phase_v1 < 3
			}
		}
		scope:ffcs_settlement_project = { set_variable = ffcs_internal_transfer_guard_v1 }
		ffcs_apply_settlement_phase_v2 = { DIVISOR = 2 }
		scope:ffcs_settlement_project = {
			remove_variable = ffcs_internal_transfer_guard_v1
			set_variable = { name = ffcs_settlement_phase_v1 value = 3 }
		}
		scope:ffcs_settlement_sponsor = {
			post_notification = ffcs_settlement_phase_3
		}
	}
	if = {
		limit = {
			scope:ffcs_settlement_project = {
				var:ffcs_settlement_progress_v1 >= 95
				var:ffcs_settlement_phase_v1 < 4
			}
		}
		scope:ffcs_settlement_project = { set_variable = ffcs_internal_transfer_guard_v1 }
		ffcs_apply_settlement_phase_v2 = { DIVISOR = 1 }
		scope:ffcs_settlement_project = {
			remove_variable = ffcs_internal_transfer_guard_v1
			set_variable = { name = ffcs_settlement_phase_v1 value = 4 }
		}
		if = {
			limit = {
				scope:ffcs_settlement_project = {
					var:ffcs_settlement_progress_v1 < 100
				}
			}
			scope:ffcs_settlement_sponsor = {
				post_notification = ffcs_settlement_phase_4
			}
		}
	}
	if = {
		limit = {
			scope:ffcs_settlement_project = {
				var:ffcs_settlement_progress_v1 >= 100
			}
		}
		ffcs_complete_settlement_project_v1 = yes
	}
}

# Current scope = state carrying a project
ffcs_advance_settlement_project_v1 = {
	save_temporary_scope_as = ffcs_settlement_project
	state_region = { save_temporary_scope_as = ffcs_settlement_state_region }
	var:ffcs_settlement_sponsor_v1 ?= {
		save_temporary_scope_as = ffcs_settlement_sponsor
	}
	var:ffcs_settlement_original_owner_v1 ?= {
		save_temporary_scope_as = ffcs_settlement_original_owner
	}
	# Recover projects whose old quoted province IDs recorded a phase without
	# actually leaving a sponsor-owned frontier.
	if = {
		limit = {
			var:ffcs_settlement_phase_v1 >= 1
			OR = {
				NOT = { has_variable_list = ffcs_settlement_provinces_v2 }
				AND = {
					has_variable_list = ffcs_settlement_provinces_v2
					NOT = {
						any_in_list = {
							variable = ffcs_settlement_provinces_v2
							state.owner = scope:ffcs_settlement_sponsor
						}
					}
				}
			}
		}
		clear_variable_list = ffcs_settlement_provinces_v2
		set_variable = { name = ffcs_settlement_phase_v1 value = 0 }
	}
	state_region = {
		every_scope_state = {
			limit = { owner = scope:ffcs_settlement_original_owner }
			save_temporary_scope_as = ffcs_settlement_target_state
		}
	}
	if = {
		limit = {
			exists = scope:ffcs_settlement_target_state
			ffcs_settlement_project_remains_valid = {
				COUNTRY = var:ffcs_settlement_sponsor_v1
				TARGET = var:ffcs_settlement_original_owner_v1
			}
		}
		debug_log = "FFCS|EVALUATION_PASSED|DATE=[TimeKeeper.GetCurrentDate.GetString]|SPONSOR=[SCOPE.sCountry('ffcs_settlement_sponsor').GetNameNoFormatting]|REGION=[SCOPE.sStateRegion('ffcs_settlement_state_region').GetNameNoFormatting]|TARGET=[SCOPE.sState('ffcs_settlement_target_state').GetNameNoFormatting]"
		scope:ffcs_settlement_target_state = {
			ffcs_apply_monthly_settlement_progress_v1 = yes
		}
	}
	else = {
		debug_log = "FFCS|EVALUATION_FAILED|DATE=[TimeKeeper.GetCurrentDate.GetString]|SPONSOR=[SCOPE.sCountry('ffcs_settlement_sponsor').GetNameNoFormatting]|REGION=[SCOPE.sStateRegion('ffcs_settlement_state_region').GetNameNoFormatting]|PROJECT=[SCOPE.sState('ffcs_settlement_project').GetNameNoFormatting]"
		ffcs_cancel_settlement_project_v1 = yes
	}
}
