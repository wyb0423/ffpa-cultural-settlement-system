# Root = country
ffcs_uses_cultural_settlement_law = {
	OR = {
		has_law_or_variant = law_type:law_colonial_resettlement
		has_law_or_variant = law_type:law_frontier_colonization
	}
}

# Root = country
ffcs_below_settlement_cap = {
	OR = {
		NOT = { has_variable = ffcs_active_settlement_count_v1 }
		var:ffcs_active_settlement_count_v1 < 2
		AND = {
			institution_investment_level = {
				institution = institution_colonial_affairs
				value >= 2
			}
			var:ffcs_active_settlement_count_v1 < 4
		}
		AND = {
			institution_investment_level = {
				institution = institution_colonial_affairs
				value >= 3
			}
			var:ffcs_active_settlement_count_v1 < 6
		}
		AND = {
			institution_investment_level = {
				institution = institution_colonial_affairs
				value >= 4
			}
			var:ffcs_active_settlement_count_v1 < 8
		}
		AND = {
			institution_investment_level = {
				institution = institution_colonial_affairs
				value >= 5
			}
			var:ffcs_active_settlement_count_v1 < 10
		}
	}
}

# Current scope = state; COUNTRY = acting country; TARGET = target owner; STATE = target state; REGION = target strategic region
ffcs_state_is_eligible_for_settlement = {
	custom_tooltip = {
		text = FFCS_CULTURAL_HOMELAND_ONLY_TT
		is_homeland_of_country_cultures = $COUNTRY$
	}
	NOT = {
		state_region = {
			OR = {
				has_variable = ffcs_settlement_sponsor_v1
				any_scope_state = { has_variable = ffcs_settlement_sponsor_v1 }
			}
		}
	}
	owner = {
		is_country_type = decentralized
		NOT = { is_diplomatic_play_enemy_of = $COUNTRY$ }
	}
	custom_tooltip = {
		text = FFCS_VALID_SETTLEMENT_ROUTE_TT
		OR = {
			ffcs_generated_has_land_seed_v2 = {
				COUNTRY = $COUNTRY$
				TARGET = $TARGET$
			}
			AND = {
				is_coastal = yes
				$COUNTRY$ = { has_strategic_adjacency = $STATE$ }
				$COUNTRY$ = { has_port_country = yes }
				ffcs_generated_has_port_seed_v2 = { TARGET = $TARGET$ }
			}
		}
	}
	OR = {
		NOT = { has_state_trait = state_trait_severe_malaria }
		$COUNTRY$ = { has_technology_researched = quinine }
	}
	$COUNTRY$ = {
		has_strategic_region_interest_tier = {
			strategic_region = $REGION$
			value >= interest_tier_effect_stake_colonial_claim_min_tier
		}
	}
}

# Current scope = state carrying a project; COUNTRY = sponsor; TARGET = original owner
ffcs_settlement_project_remains_valid = {
	has_variable = ffcs_settlement_sponsor_v1
	has_variable = ffcs_settlement_original_owner_v1
	has_variable = ffcs_settlement_route_v2
	OR = {
		owner = $COUNTRY$
		owner = $TARGET$
	}
	state_region = {
		any_scope_state = { owner = $TARGET$ }
	}
	is_homeland_of_country_cultures = $COUNTRY$
	OR = {
		AND = {
			NOT = { has_variable_list = ffcs_settlement_provinces_v2 }
			OR = {
				ffcs_generated_has_land_seed_v2 = {
					COUNTRY = $COUNTRY$
					TARGET = $TARGET$
				}
				AND = {
					$COUNTRY$ = { has_port_country = yes }
					is_coastal = yes
					ffcs_generated_has_port_seed_v2 = {
						TARGET = $TARGET$
					}
				}
			}
		}
		AND = {
			has_variable_list = ffcs_settlement_provinces_v2
			var:ffcs_settlement_route_v2 = 1
			any_in_list = {
				variable = ffcs_settlement_provinces_v2
				state.owner = $COUNTRY$
			}
		}
		AND = {
			has_variable_list = ffcs_settlement_provinces_v2
			var:ffcs_settlement_route_v2 = 2
			$COUNTRY$ = { has_port_country = yes }
			ffcs_generated_country_owns_port_v2 = {
				COUNTRY = $COUNTRY$
			}
			any_in_list = {
				variable = ffcs_settlement_provinces_v2
				state.owner = $COUNTRY$
			}
			state_region = {
				any_scope_state = {
					owner = $COUNTRY$
					has_building = building_port
				}
			}
		}
	}
	$COUNTRY$ = {
		ffcs_uses_cultural_settlement_law = yes
		has_technology_researched = colonization
		OR = {
			is_country_type = recognized
			is_country_type = unrecognized
		}
		NOT = { is_diplomatic_play_enemy_of = $TARGET$ }
	}
	$TARGET$ = { is_country_type = decentralized }
}
