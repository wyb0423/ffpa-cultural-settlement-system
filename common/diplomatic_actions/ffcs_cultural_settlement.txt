da_ffcs_establish_cultural_settlement = {
	groups = { general }

	requires_approval = no
	should_notify_third_parties = yes
	show_effect_in_tooltip = no
	state_selection = second_required
	second_state_list = second_country

	selectable = {
		OR = {
			is_country_type = recognized
			is_country_type = unrecognized
		}
		has_technology_researched = colonization
	}

	potential = {
		NOT = { scope:target_country = root }
		scope:target_country = { is_country_type = decentralized }
	}

	possible = {
		custom_tooltip = {
			text = FFCS_REQUIRES_CULTURAL_SETTLEMENT_LAW_TT
			ffcs_uses_cultural_settlement_law = yes
		}
		is_at_war = no
		in_default = no
		ffcs_below_settlement_cap = yes
	}

	second_state_trigger = {
		ffcs_state_is_eligible_for_settlement = {
			COUNTRY = scope:country
			TARGET = scope:target_country
			STATE = root
			REGION = root.region
		}
	}

	accept_effect = {
		if = {
			limit = { exists = scope:second_state }
			save_temporary_scope_as = ffcs_settlement_sponsor

			ffcs_initialize_country_schema_v1 = yes
			if = {
				limit = { NOT = { has_variable = ffcs_active_settlement_count_v1 } }
				set_variable = { name = ffcs_active_settlement_count_v1 value = 0 }
			}
			change_variable = {
				name = ffcs_active_settlement_count_v1
				add = 1
			}
			scope:target_country = {
				save_temporary_scope_as = ffcs_settlement_original_owner
				ffcs_initialize_country_schema_v1 = yes
			}

			scope:second_state = {
				save_temporary_scope_as = ffcs_settlement_project
				state_region = {
					save_temporary_scope_as = ffcs_settlement_state_region
					set_variable = {
						name = ffcs_settlement_sponsor_v1
						value = scope:ffcs_settlement_sponsor
					}
				}
				add_modifier = { name = ffcs_cultural_settlement_project }
				if = {
					limit = {
						ffcs_generated_has_land_seed_v2 = {
							COUNTRY = scope:ffcs_settlement_sponsor
							TARGET = scope:ffcs_settlement_original_owner
						}
					}
					set_variable = { name = ffcs_settlement_route_v2 value = 1 }
				}
				else_if = {
					limit = {
						ffcs_generated_has_port_seed_v2 = {
							TARGET = scope:ffcs_settlement_original_owner
						}
					}
					set_variable = { name = ffcs_settlement_route_v2 value = 2 }
				}
				set_variable = {
					name = ffcs_settlement_sponsor_v1
					value = scope:ffcs_settlement_sponsor
				}
				set_variable = {
					name = ffcs_settlement_original_owner_v1
					value = scope:ffcs_settlement_original_owner
				}
				set_variable = { name = ffcs_settlement_progress_v1 value = 0 }
				set_variable = { name = ffcs_settlement_phase_v1 value = 0 }
				set_variable = {
					name = ffcs_settlement_resistance_v1
					value = {
						add = 10
						add = {
							value = state_population
							divide = 100000
							max = 30
						}
					}
				}
				owner = {
					if = {
						limit = { NOT = { has_variable = ffcs_inbound_settlement_count_v1 } }
						set_variable = { name = ffcs_inbound_settlement_count_v1 value = 0 }
					}
					change_variable = {
						name = ffcs_inbound_settlement_count_v1
						add = 1
					}
				}
				scope:ffcs_settlement_sponsor = {
					add_to_variable_list = {
						name = ffcs_active_settlement_states_v1
						target = scope:ffcs_settlement_project
					}
					if = {
						limit = {
							NOT = {
								has_journal_entry = je_ffcs_cultural_settlement_overview
							}
						}
						add_journal_entry = {
							type = je_ffcs_cultural_settlement_overview
						}
					}
				}
				debug_log = "FFCS|PROJECT_CREATED|DATE=[TimeKeeper.GetCurrentDate.GetString]|SPONSOR=[SCOPE.sCountry('ffcs_settlement_sponsor').GetNameNoFormatting]|REGION=[SCOPE.sStateRegion('ffcs_settlement_state_region').GetNameNoFormatting]|PROJECT=[SCOPE.sState('ffcs_settlement_project').GetNameNoFormatting]|LOCK=SET"
				if = {
					limit = { var:ffcs_settlement_route_v2 = 2 }
					scope:ffcs_settlement_sponsor = { add_treasury = -100000 }
				}
				if = {
					limit = { num_provinces = 1 }
					set_variable = { name = ffcs_settlement_progress_v1 value = 100 }
					ffcs_complete_settlement_project_v1 = yes
				}
				else = {
					scope:ffcs_settlement_project = {
						set_variable = ffcs_internal_transfer_guard_v1
					}
					ffcs_apply_settlement_phase_v2 = { DIVISOR = 4 }
					scope:ffcs_settlement_project = {
						remove_variable = ffcs_internal_transfer_guard_v1
					}
					if = {
						limit = {
							scope:ffcs_settlement_project = {
								has_variable_list = ffcs_settlement_provinces_v2
							}
						}
						scope:ffcs_settlement_project = {
							set_variable = { name = ffcs_settlement_progress_v1 value = 25 }
							set_variable = { name = ffcs_settlement_phase_v1 value = 1 }
						}
					}
				}
			}

			create_incident = {
				country = scope:target_country
				value = 5
			}
		}
		else = {
			custom_tooltip = FFCS_SELECT_TARGET_STATE_TT
		}
	}

	is_hostile = yes

	ai = {
		evaluation_chance = {
			value = 0
			if = {
				limit = {
					ffcs_uses_cultural_settlement_law = yes
					has_technology_researched = colonization
					in_default = no
					is_at_war = no
					ffcs_below_settlement_cap = yes
				}
				# Firefall starts with many minor polities; give every otherwise eligible
				# country the native major-power consideration chance.
				add = 0.05
				if = {
					limit = { country_rank = rank_value:great_power }
					add = 0.05
				}
				if = {
					limit = { country_is_colonial_or_company = yes }
					add = 0.05
				}
			}
		}

		will_select_as_second_state = {
			always = yes
		}

		will_propose_with_states = {
			exists = scope:second_state
			scope:second_state = {
				ffcs_state_is_eligible_for_settlement = {
					COUNTRY = root
					TARGET = scope:target_country
					STATE = scope:second_state
					REGION = scope:second_state.region
				}
			}
		}

		propose_score = {
			value = 10

			if = {
				limit = { is_adjacent_to_state = scope:second_state }
				add = 50
			}
			else = {
				multiply = 0.25
			}

			if = {
				limit = { scope:second_state = { has_claim_by = root } }
				add = 40
			}
			if = {
				limit = {
					scope:second_state.region = {
						any_scope_state = { owner = root }
					}
				}
				add = 30
			}
			if = {
				limit = { scope:second_state = { is_coastal = yes } }
				add = 10
			}
		}

		will_propose = {
			always = yes
		}
	}
}
